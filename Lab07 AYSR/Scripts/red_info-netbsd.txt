rm /root/red_info.sh
cat > /root/red_info.sh << 'EOF'
#!/bin/sh

while true; do
    clear
    echo ""
    echo "  INFORMACION DE RED - NetBSD"
    echo ""
    echo "  1. Conexiones activas"
    echo "  2. Tabla de enrutamiento"
    echo "  3. Estadisticas de interfaces"
    echo "  4. Puertos en escucha"
    echo "  5. Estadisticas de protocolos"
    echo "  6. Salir"
    echo ""
    printf "  Seleccione una opcion: "
    read opcion

    case $opcion in
        1)
            clear
            echo ""
            echo "  Conexiones activas en este momento"
            echo ""
            netstat -an | grep ESTABLISHED
            ;;
        2)
            clear
            echo ""
            echo "  Tabla de enrutamiento del sistema"
            echo ""
            route -n show
            ;;
        3)
            clear
            echo ""
            echo "  Estadisticas de las interfaces de red"
            echo ""
            netstat -i
            ;;
        4)
            clear
            echo ""
            echo "  Puertos que estan escuchando conexiones"
            echo ""
            netstat -an | grep LISTEN
            ;;
        5)
            clear
            echo ""
            echo "  Estadisticas de protocolos de red"
            echo ""
            netstat -s | head -40
            ;;
        6)
            clear
            exit 0
            ;;
        *)
            echo ""
            echo "  Opcion no valida, intente de nuevo"
            ;;
    esac
    echo ""
    printf "  Presione Enter para continuar"
    read dummy
done
EOF
