while ($true) {
    Clear-Host
    Write-Host ""
    Write-Host "  INFORMACION DE RED - Windows"
    Write-Host ""
    Write-Host "  1. Conexiones activas"
    Write-Host "  2. Tabla de enrutamiento"
    Write-Host "  3. Estadisticas de interfaces"
    Write-Host "  4. Puertos en escucha"
    Write-Host "  5. Estadisticas de protocolos"
    Write-Host "  6. Salir"
    Write-Host ""
    $opcion = Read-Host "  Seleccione una opcion"

    switch ($opcion) {
        "1" {
            Clear-Host
            Write-Host ""
            Write-Host "  Conexiones activas en este momento"
            Write-Host ""
            netstat -an | Select-String "ESTABLISHED"
        }
        "2" {
            Clear-Host
            Write-Host ""
            Write-Host "  Tabla de enrutamiento del sistema"
            Write-Host ""
            Get-NetRoute | Format-Table DestinationPrefix, NextHop, RouteMetric, ifIndex
        }
        "3" {
            Clear-Host
            Write-Host ""
            Write-Host "  Estadisticas de las interfaces de red"
            Write-Host ""
            Get-NetAdapter | Format-Table Name, Status, LinkSpeed, MacAddress
        }
        "4" {
            Clear-Host
            Write-Host ""
            Write-Host "  Puertos que estan escuchando conexiones"
            Write-Host ""
            netstat -an | Select-String "LISTENING"
        }
        "5" {
            Clear-Host
            Write-Host ""
            Write-Host "  Estadisticas de protocolos de red"
            Write-Host ""
            netstat -s
        }
        "6" {
            Clear-Host
            exit
        }
        default {
            Write-Host ""
            Write-Host "  Opcion no valida, intente de nuevo"
        }
    }
    Write-Host ""
    Read-Host "  Presione Enter para continuar"
}
