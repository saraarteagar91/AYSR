cat > /root/verificar_puerto.sh << 'EOF'
#!/bin/bash

clear
echo ""
echo "  VERIFICADOR DE PUERTOS - Slackware"
echo ""
printf "  Ingrese el puerto que desea verificar: "
read puerto

resultado=$(netstat -an | grep ":$puerto " | head -1)

if [ -z "$resultado" ]; then
    echo ""
    echo "  El puerto $puerto esta CERRADO"
    echo ""
else
    estado=$(echo $resultado | awk '{print $6}')
    proto=$(echo $resultado | awk '{print $1}')
    echo ""
    echo "  El puerto $puerto esta ABIERTO"
    echo ""
    echo "  Protocolo: $proto"
    echo "  Estado: $estado"
    echo ""

    case $puerto in
        80)    echo "  Servicio: HTTP - Servidor web" ;;
        443)   echo "  Servicio: HTTPS - Servidor web seguro" ;;
        22)    echo "  Servicio: SSH - Acceso remoto seguro" ;;
        21)    echo "  Servicio: FTP - Transferencia de archivos" ;;
        161)   echo "  Servicio: SNMP - Monitoreo de red" ;;
        10050) echo "  Servicio: Zabbix Agent" ;;
        10051) echo "  Servicio: Zabbix Server" ;;
        5432)  echo "  Servicio: PostgreSQL - Base de datos" ;;
        *)     echo "  Servicio: Desconocido o personalizado" ;;
    esac
fi
echo ""
EOF
